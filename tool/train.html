def norm(V):
    Vmin = V.min()
    Vmax = V.max()

    return (V - Vmin) / (Vmax - Vmin), Vmin, Vmax


def make_data(X, Y, ptest, batch):
    Xnorm, Xmin, Xmax = norm(X)

    rng = np.random.default_rng(seed)
    idXrand = rng.permutation(len(Xnorm))

    ntest = int(round(ptest / 100.0 * len(Xnorm)))

    idXtest = idXrand[:ntest]
    idXtrain = idXrand[ntest:]

    Xtrain = torch.tensor(Xnorm.iloc[idXtrain].to_numpy(dtype=np.float32))
    Xtest = torch.tensor(Xnorm.iloc[idXtest].to_numpy(dtype=np.float32))

    Ytrain = torch.tensor(Y.iloc[idXtrain].to_numpy(dtype=np.float32))
    Ytest = torch.tensor(Y.iloc[idXtest].to_numpy(dtype=np.float32))

    trainloader = DataLoader(TensorDataset(Xtrain, Ytrain), batch_size=batch, shuffle=True)
    testloader = DataLoader(TensorDataset(Xtest, Ytest), batch_size=batch, shuffle=False)

    print("device =", device)

    print("\nXmin")
    display(Xmin)

    print("\nXmax")
    display(Xmax)

    print("train  =", Xtrain.shape, Ytrain.shape)
    print("test   =", Xtest.shape, Ytest.shape)

    return Xnorm, Xmin, Xmax, Xtrain, Xtest, Ytrain, Ytest, trainloader, testloader

class FTTransformer(nn.Module):
    def __init__(self, nfeature, dmodel=32, nhead=4, nlayer=8, dropout=0.1):
        super().__init__()

        self.weight = nn.Parameter(torch.randn(nfeature, dmodel) * 0.02)
        self.bias = nn.Parameter(torch.zeros(nfeature, dmodel))

        layer = nn.TransformerEncoderLayer(
            d_model=dmodel,
            nhead=nhead,
            dim_feedforward=4 * dmodel,
            dropout=dropout,
            activation="gelu",
            # >> New PyTorch
            batch_first=True,
            norm_first=True,
            # <<
        )

        self.encoder = nn.TransformerEncoder(layer, num_layers=nlayer)

        self.head = nn.Sequential(
            nn.LayerNorm(dmodel),
            nn.Linear(dmodel, dmodel),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(dmodel, 1),
        )

    def forward(self, X):
        H = X.unsqueeze(-1) * self.weight.unsqueeze(0) + self.bias.unsqueeze(0)

        # >> New PyTorch
        H = self.encoder(H)
        # <<

        # >> Old PyTorch
        # H = H.transpose(0, 1)
        # H = self.encoder(H)
        # H = H.transpose(0, 1)
        # <<

        H = H.mean(dim=1)

        return self.head(H)

class MLP(nn.Module):
    def __init__(self, nfeature, dropout=0.1):
        super().__init__()

        self.net = nn.Sequential(
            nn.Linear(nfeature, 64),
            nn.GELU(),
            nn.Dropout(dropout),

            nn.Linear(64, 64),
            nn.GELU(),
            nn.Dropout(dropout),

            nn.Linear(64, 32),
            nn.GELU(),
            nn.Dropout(dropout),

            nn.Linear(32, 16),
            nn.GELU(),
            nn.Dropout(dropout),

            nn.Linear(16, 8),
            nn.GELU(),
            nn.Dropout(dropout),

            nn.Linear(8, 1),
        )

    def forward(self, X):
        return self.net(X)

def run(model, name, nepoch, lr, weight_decay):
    model = model.to(device)

    criterion = nn.SmoothL1Loss()
    optimizer = torch.optim.AdamW(
        model.parameters(),
        lr=lr,
        weight_decay=weight_decay,
    )

    history = {
        "train_loss": [],
        "test_loss": [],
        "train_mae": [],
        "test_mae": [],
        "train_mrae": [],
        "test_mrae": [],
    }

    progress = tqdm(range(1, nepoch + 1), desc=name)

    for epoch in progress:

        model.train()
        train_loss = 0.0
        train_mae = 0.0
        train_mrae = 0.0
        train_total = 0

        for Xb, Yb in trainloader:
            Xb = Xb.to(device)
            Yb = Yb.to(device)

            optimizer.zero_grad()

            Ybpred = model(Xb)
            loss = criterion(Ybpred, Yb)

            loss.backward()
            optimizer.step()

            n = Xb.shape[0]

            train_loss += loss.item() * n
            train_mae += torch.abs(Ybpred - Yb).sum().item()
            train_mrae += torch.abs((Ybpred - Yb) / Yb).sum().item()
            train_total += n

        train_loss /= train_total
        train_mae /= train_total
        train_mrae = 100.0 * train_mrae / train_total


        model.eval()
        test_loss = 0.0
        test_mae = 0.0
        test_mrae = 0.0
        test_total = 0

        with torch.no_grad():
            for Xb, Yb in testloader:
                Xb = Xb.to(device)
                Yb = Yb.to(device)

                Ybpred = model(Xb)
                loss = criterion(Ybpred, Yb)

                n = Xb.shape[0]

                test_loss += loss.item() * n
                test_mae += torch.abs(Ybpred - Yb).sum().item()
                test_mrae += torch.abs((Ybpred - Yb) / Yb).sum().item()
                test_total += n

        test_loss /= test_total
        test_mae /= test_total
        test_mrae = 100.0 * test_mrae / test_total

        history["train_loss"].append(train_loss)
        history["test_loss"].append(test_loss)
        history["train_mae"].append(train_mae)
        history["test_mae"].append(test_mae)
        history["train_mrae"].append(train_mrae)
        history["test_mrae"].append(test_mrae)

        progress.set_postfix(
            train_loss=f"{train_loss:.4e}",
            test_loss=f"{test_loss:.4e}",
            train_mae=f"{train_mae:.4e}",
            test_mae=f"{test_mae:.4e}",
            train_mrae=f"{train_mrae:.4e}%",
            test_mrae=f"{test_mrae:.4e}%",
        )

    return history

def predict(model, loader):
    model.eval()

    Vpred = []
    Vtrue = []

    with torch.no_grad():
        for Xb, Yb in loader:
            Xb = Xb.to(device)

            Ybpred = model(Xb)

            Vpred.append(Ybpred.cpu().numpy())
            Vtrue.append(Yb.numpy())

    Ypred = np.concatenate(Vpred).reshape(-1)
    Ytrue = np.concatenate(Vtrue).reshape(-1)

    return Ypred, Ytrue
